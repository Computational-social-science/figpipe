create table mag_staging.project_phoenix_text_data 
	WITH (
		format = 'TEXTFILE',
		field_delimiter = '---!!*%^&%^&*!!---',
		external_location = 's3://mag-output/PROJECT_Phoenix/Text_Data/'
	  )
	as 

with final_table as (Select 
			REPLACE(REPLACE(txt.concept_id,'https://openalex.org/',''),'c','C') as concept_id,
			txt.work_id,
			txt.year,
			code.country,
			txt.title,
			REPLACE(REPLACE(txt.abstract, '\t', ''),'\n','') AS abstract
FROM (
		SELECT DISTINCT concept_ids.concept_id,
			concept_ids.work_id,
			abstract.year,
			abstract.title,
			abstract.abstract
		FROM (
				Select CIDS.concept_id,
					WCIDS.work_id
				FROM "open_alex"."works_concepts" WCIDS
					JOIN "open_alex"."concepts_ids" as CIDS ON WCIDS.concept_id = CIDS.concept_ID

					WHERE cast(CIDS.score as double) > 0
			) as concept_ids
			JOIN (
				SELECT DISTINCT id,
					title,
					abstract,
					year
				FROM "open_alex"."semantic_works"
				WHERE type = 'article'
					AND abstract <> ''
			) as abstract ON abstract.id = concept_ids.work_id
	) as txt
	JOIN (
		Select work_id,
			array_join(array_agg(concat(country_code,'+',author_position)), ' ') as country
		From (
				SELECT work_id,
					institution_id,
					author_position
				FROM "open_alex"."works_authorships"
			) auth
			join(
				SELECT id,
					country_code
				FROM "open_alex"."institutions"
			) inst ON inst.id = auth.institution_id
		group by work_id
	) as code ON code.work_id = txt.work_id)
Select * from final_table
where concept_id <> ''
