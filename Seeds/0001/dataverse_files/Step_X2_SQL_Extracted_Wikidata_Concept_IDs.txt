with term_phrases_concepts as (with origin_papers as (
	with firstpaper_unfiltered as (
		with papers as (
			SELECT id,
				year
			FROM "open_alex"."semantic_works"
		)
		select id,
			works_concepts.concept_id,
			wikidata,
			display_name,
			concept_level,
			year,
			row_number() over (
				partition by works_concepts.concept_id
				Order by year
			) as rownumber
		from works_concepts
			JOIN papers ON papers.id = works_concepts.work_id
			JOIN (
				Select id as concept_id,
					wikidata,
					display_name,
					level as concept_level
				from "open_alex"."concepts"
			) as concept_level on concept_level.concept_id = works_concepts.concept_id
	)
	select id,
		concept_id,
		wikidata,
		display_name,
		concept_level,
		year
	from firstpaper_unfiltered
	where rownumber = 1
		and concept_level >= '2'
),
country_of_origin as (
	SELECT work_id,
		array_join(array_agg(country_code), ' ') as country_code
	FROM "open_alex"."works_authorships"
		JOIN (
			SELECT id,
				country_code
			FROM "open_alex"."institutions"
		) as institutions ON institutions.id = works_authorships.institution_id
	group by work_id
)
Select id as origin_work_id,
	origin_papers.concept_id,
	concat(publication_year,'+',works_concepts.work_id) as work_year,
	wikidata,
	display_name,
	year as origin_year,
	concept_level,
	country_code as origin_country
from origin_papers
	join country_of_origin on country_of_origin.work_id = origin_papers.id
	join works_concepts on works_concepts.concept_id = origin_papers.concept_id
	join (select id as work_id, publication_year from "open_alex"."works") work_year on work_year.work_id =  	works_concepts.work_id
where year >= '1990')
select concept_id, wikidata, display_name, origin_work_id, origin_country, origin_year, array_join(array_agg(work_year), ',') as work_id
from term_phrases_concepts
group by concept_id, wikidata, display_name, origin_work_id, origin_country, origin_year
